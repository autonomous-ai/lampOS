# /etc/systemd/system/lamp-memory.service
# lampOS memory: what the lamp keeps from one day to the next (lampOS/docs/memory.md). Its own user: no other part reads its records.
# The product unit (lampOS/docs/install.md).
[Unit]
Description=lampOS memory (lamp-memory)
PartOf=lampos.target
# Not before setup has finished (#806). The target's own condition holds back none of its parts.
ConditionPathExists=/var/lib/lampos-setup/done
After=local-fs.target
StartLimitIntervalSec=600
StartLimitBurst=5

[Service]
Type=simple
User=lampos-memory
Group=lampos-memory
Slice=lampos-parts.slice
StateDirectory=lampos-memory
StateDirectoryMode=0700
RuntimeDirectory=lampos-memory
RuntimeDirectoryMode=0755
PrivateNetwork=yes
SupplementaryGroups=lampos
# --zone system: the lamp's zone from /etc/localtime, which setup points at the person's (#820), across summer time.
ExecStart=/opt/lampOS/memory/current/lamp-memory --dir /var/lib/lampos-memory --socket /run/lampos-memory/memory.sock --keep-days 30 --zone system --memory-days --bus-root /run/lampos-bus
Restart=on-failure
RestartSec=2s
StandardInput=null
StandardOutput=journal
StandardError=journal
SyslogIdentifier=lamp-memory
NoNewPrivileges=yes
ProtectSystem=strict
ReadWritePaths=/run/lampos-bus
ProtectHome=yes
PrivateTmp=yes

[Install]
WantedBy=lampos.target

# lampOS voice: listens, talks with the model, speaks and drives the light ring (lampOS/docs/serve-runtime.md).
# The product unit (lampOS/docs/install.md): every lamp's, with the office lamp's options and switches of
# 2026-10-17 (<redacted>'s unit that day, without what only the office has). A lamp that needs a switch off says so in lamp-voice.service.d/, never here.
# Root: it reads /etc/lampOS/voice (the credential) and runs its microphone and speaker real-time.
[Unit]
Description=lampOS voice (lamp-voice serve)
PartOf=lampos.target
# Not before setup has finished (#806). The target's own condition holds back none of its parts.
ConditionPathExists=/var/lib/lampos-setup/done
# Memory, actions, schedule and tasks answer its calls from the first turn: all first.
After=network-online.target sound.target lamp-memory.service lamp-actions.service lamp-schedule.service lamp-tasks.service
# The image's speaker level, set by udev when the C-Media card appears (lampOS/docs/install.md): after it, so the
# first word is at its level. Not Wanted: a lamp without that card or unit starts the voice all the same.
After=lamp-cmedia-speaker.service
Wants=network-online.target
StartLimitIntervalSec=600
StartLimitBurst=10
